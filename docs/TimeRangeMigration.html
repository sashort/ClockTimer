TimeRange Migration

This document defines the post-migration responsibilities and provides copyable JavaScript fragments. Each Copy button copies only the JavaScript in its associated block.

Migration contract

TimeRange owns its individual interval and counted state.

TimeRangeGroup owns aggregate counted time and emits a boundary event when a boundary is reached.

ClockTimer owns timer lifecycle/orchestration and reacts to group events.

RingContainer owns presentation.

The running application now uses TimeRangeModel.js and its TimeRangeElement.js DOM adapter. The former element and counted-group implementations have been retired. ClockTimer consumes tick-driven boundary batches and model-owned counted totals. See the current model contract for update transactions, split points, identity, presentation notifications, and the batched event format. The fragments below illustrate the original migration responsibilities.

TimeRangeGroup — aggregate counted time

class TimeRangeGroup {
    constructor(timeRanges = []) {
        this.timeRanges = [];
        this.totalCountedTime = 0;
        this.elapsedCountedTime = 0;
        this.remainingCountedTime = 0;
        this._listeners = new Set();
        this.addAll(timeRanges);
    }

    addEventListener(type, listener) {
        if (type === "boundary-reached") this._listeners.add(listener);
    }

    removeEventListener(type, listener) {
        if (type === "boundary-reached") this._listeners.delete(listener);
    }

    _emitBoundary(before, after) {
        const event = { type: "boundary-reached" };
        if (before) event.before = before;
        if (after) event.after = after;
        for (const listener of this._listeners) listener(event);
    }

    add(timeRange) {
        if (!timeRange || this.timeRanges.includes(timeRange)) return;
        this.timeRanges.push(timeRange);
        this.recalculateCountedTime();
    }

    addAll(timeRanges) {
        for (const timeRange of timeRanges) this.add(timeRange);
    }

    remove(timeRange) {
        const index = this.timeRanges.indexOf(timeRange);
        if (index !== -1) {
            this.timeRanges.splice(index, 1);
            this.recalculateCountedTime();
        }
    }

    clear() {
        this.timeRanges.length = 0;
        this.recalculateCountedTime();
    }

    recalculateCountedTime() {
        this.totalCountedTime = this.timeRanges.reduce(
            (sum, range) => sum + Number(range.counted?.total || 0), 0
        );
        this.elapsedCountedTime = this.timeRanges.reduce(
            (sum, range) => sum + Number(range.counted?.elapsed || 0), 0
        );
        this.remainingCountedTime = this.timeRanges.reduce(
            (sum, range) => sum + Number(range.counted?.remaining || 0), 0
        );
        return this;
    }
}
